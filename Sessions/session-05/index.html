🆘 پروژه ۳: پیام SOS با مورس

پروژه ۳ — متوسط

پیام نجات با کد مورس

پیام SOS (... --- ...) را با LED ارسال می‌کنیم. این پروژه به ما یاد می‌دهد که چگونه از توابع برای جلوگیری از تکرار کد استفاده کنیم.

🎯 هدف:
• تعریف توابع سفارشی
• الگوهای زمانی پیچیده
• مفهوم کد مورس
باز کردن Wokwi 
/*
  پروژه ۳: پیام SOS با کد مورس
  S = ...   O = ---
*/

int ledPin = 13;

void setup() {
  pinMode(ledPin, OUTPUT);
}

// تابع نقطه (کوتاه)
void dot() {
  digitalWrite(ledPin, HIGH);
  delay(200);
  digitalWrite(ledPin, LOW);
  delay(200);
}

// تابع خط (بلند)
void dash() {
  digitalWrite(ledPin, HIGH);
  delay(600);
  digitalWrite(ledPin, LOW);
  delay(200);
}

void loop() {
  // حرف S: ...
  dot(); dot(); dot();
  delay(400);
  
  // حرف O: ---
  dash(); dash(); dash();
  delay(400);
  
  // حرف S: ...
  dot(); dot(); dot();
  
  // توقف طولانی قبل از تکرار
  delay(2000);
}

🌊 پروژه ۴: Fade (محو تدریجی)

پروژه ۴ — متوسط

محو تدریجی LED

با استفاده از PWM (Pulse Width Modulation)، روشنایی LED را به‌صورت تدریجی کم و زیاد می‌کنیم.

🎯 هدف:
• آشنایی با PWM
• استفاده از حلقه for
• درک analogWrite
• پین‌های PWM (با ~)

⚠️ نکته: PWM فقط روی پین‌های ۳، ۵، ۶، ۹، ۱۰، ۱۱ کار می‌کند.

باز کردن Wokwi 
/*
  پروژه ۴: Fade — محو تدریجی LED
  با استفاده از PWM
*/

int ledPin = 9;  // پین PWM

void setup() {
  pinMode(ledPin, OUTPUT);
}

void loop() {
  // زیاد کردن روشنایی از ۰ تا ۲۵۵
  for (int brightness = 0; brightness <= 255; brightness += 5) {
    analogWrite(ledPin, brightness);
    delay(30);
  }
  
  // کم کردن روشنایی از ۲۵۵ تا ۰
  for (int brightness = 255; brightness >= 0; brightness -= 5) {
    analogWrite(ledPin, brightness);
    delay(30);
  }
}

🔘 پروژه ۵: کنترل با دکمه

پروژه ۵ — متوسط

کنترل LED با دکمه

با فشار دادن دکمه، LED روشن و با رها کردن، خاموش می‌شود. این پروژه اولین ورودی دیجیتال ماست.

🎯 هدف:
• آشنایی با digitalRead
• تنظیم پین به INPUT
• شرط if
• مقاومت Pull-up
باز کردن Wokwi 
/*
  پروژه ۵: کنترل LED با دکمه
  فشار دهید → LED روشن
  رها کنید  → LED خاموش
*/

int buttonPin = 2;
int ledPin = 13;

void setup() {
  pinMode(ledPin, OUTPUT);
  pinMode(buttonPin, INPUT_PULLUP);  // مقاومت داخلی Pull-up
}

void loop() {
  int buttonState = digitalRead(buttonPin);
  
  // با INPUT_PULLUP، دکمه فشار داده‌نشده = HIGH
  if (buttonState == LOW) {
    digitalWrite(ledPin, HIGH);
  } else {
    digitalWrite(ledPin, LOW);
  }
}

🔀 پروژه ۶: سوئیچ تاگل

پروژه ۶ — پیشرفته

سوئیچ روشن/خاموش با یک دکمه

با هر فشار دکمه، LED را روشن یا خاموش می‌کنیم (مثل کلید برق). این پروژه مفهوم State Machine را معرفی می‌کند.

🎯 هدف:
• تشخیص لبه (Edge Detection)
• متغیر وضعیت (State)
• جلوگیری از تشخیص تکراری فشار
باز کردن Wokwi 
/*
  پروژه ۶: سوئیچ تاگل با یک دکمه
  هر فشار → تغییر وضعیت LED
*/

int buttonPin = 2;
int ledPin = 13;

bool ledState = false;      // وضعیت LED
int lastButtonState = HIGH; // وضعیت قبلی دکمه

void setup() {
  pinMode(ledPin, OUTPUT);
  pinMode(buttonPin, INPUT_PULLUP);
  digitalWrite(ledPin, ledState);
}

void loop() {
  int currentButtonState = digitalRead(buttonPin);
  
  // تشخیص لبه نزولی (از HIGH به LOW)
  if (lastButtonState == HIGH && currentButtonState == LOW) {
    ledState = !ledState;  // تغییر وضعیت
    digitalWrite(ledPin, ledState);
    delay(50);  // Debounce
  }
  
  lastButtonState = currentButtonState;
}

🫁 پروژه ۷: چراغ نفس‌کش

پروژه ۷ — پیشرفته

LED با الگوی تنفس

روشنایی LED را با یک تابع سینوسی تغییر می‌دهیم تا حس «نفس کشیدن» داشته باشد. این پروژه کاربرد ریاضیات در Arduino را نشان می‌دهد.

🎯 هدف:
• تابع sin() و cos() در Arduino
• تبدیل ریاضی به PWM
• تولید اثر نرم و طبیعی
باز کردن Wokwi 
/*
  پروژه ۷: چراغ نفس‌کش
  روشنایی با موج سینوسی تغییر می‌کند
*/

int ledPin = 9;
float angle = 0;

void setup() {
  pinMode(ledPin, OUTPUT);
}

void loop() {
  // محاسبه سینوس (بین ۰ تا ۲۵۵)
  int brightness = (sin(angle) + 1) * 127.5;
  
  analogWrite(ledPin, brightness);
  
  angle += 0.05;  // سرعت تغییر
  
  if (angle > 2 * PI) {
    angle = 0;
  }
  
  delay(20);
}

📊 پروژه ۹: نوار LED (LED Bar)

پروژه ۹ — پیشرفته

نوار پیشرفت با ۵ LED

۵ LED را به‌صورت پله‌ای روشن می‌کنیم، شبیه نوار پیشرفت (Progress Bar). این پروژه مفهوم Array و حلقه را آموزش می‌دهد.

🎯 هدف:
• استفاده از آرایه (Array)
• حلقه for روی آرایه
• ترتیب روشن شدن
باز کردن Wokwi 
/*
  پروژه ۹: نوار LED (LED Bar)
  ۵ LED به صورت پله‌ای
*/

int ledPins[] = {13, 12, 11, 10, 9};
int numLeds = 5;

void setup() {
  for (int i = 0; i < numLeds; i++) {
    pinMode(ledPins[i], OUTPUT);
  }
}

void loop() {
  // روشن کردن پله‌ای
  for (int i = 0; i < numLeds; i++) {
    digitalWrite(ledPins[i], HIGH);
    delay(200);
  }
  delay(500);
  
  // خاموش کردن پله‌ای از آخر
  for (int i = numLeds - 1; i >= 0; i--) {
    digitalWrite(ledPins[i], LOW);

🌈 پروژه ۱۰: نئوپیکسل RGB

پروژه ۱۰ — پیشرفته

LED RGB با تغییر رنگ

با استفاده از یک LED RGB، رنگ‌های مختلف را نمایش می‌دهیم. این پروژه کاربرد PWM را در سه کانال همزمان نشان می‌دهد.

🎯 هدف:
• ترکیب سه PWM (قرمز، سبز، آبی)
• تولید رنگ‌های مختلف
• مفهوم RGB در نمایشگرها
باز کردن Wokwi 
/*
  پروژه ۱۰: RGB LED — تغییر رنگ
  سه کانال: قرمز، سبز، آبی
*/

int redPin = 9;
int greenPin = 10;
int bluePin = 11;

void setup() {
  pinMode(redPin, OUTPUT);
  pinMode(greenPin, OUTPUT);
  pinMode(bluePin, OUTPUT);
}

void setColor(int r, int g, int b) {
  analogWrite(redPin, r);
  analogWrite(greenPin, g);
  analogWrite(bluePin, b);
}

void loop() {
  setColor(255, 0, 0);    // قرمز
  delay(1000);
  setColor(0, 255, 0);    // سبز
  delay(1000);
  setColor(0, 0, 255);    // آبی
  delay(1000);
  setColor(255, 255, 0);  // زرد
  delay(1000);
  setColor(255, 0, 255);  // سرخابی
  delay(1000);
  setColor(0, 255, 255);  // فیروزه‌ای
  delay(1000);
  setColor(255, 255, 255); // سفید
  delay(1000);
}

۱۰ پروژه Wokwi
  • ۱پروژه ۱ — Blink: کد را در Wokwi پیست کنید و اجرا کنید. delay را از ۱۰۰۰ به ۲۰۰ تغییر دهید.
  • ۲پروژه ۲ — چراغ راهنما: سه LED بسازید و توالی چراغ راهنما را پیاده‌سازی کنید.
  • ۳پروژه ۳ — SOS: پیام SOS را با LED ارسال کنید.
  • ۴پروژه ۴ — Fade: با PWM، روشنایی LED را تدریجی تغییر دهید.
  • ۵پروژه ۵ — دکمه: با دکمه، LED را کنترل کنید.
  • ۶پروژه ۶ — Toggle: با یک دکمه، LED را روشن/خاموش کنید.
  • ۷پروژه ۷ — نفس‌کش: الگوی تنفس را پیاده‌سازی کنید.
  • ۸پروژه ۸ — آژیر: دو LED قرمز و آبی را متناوب کنید.
  • ۹پروژه ۹ — LED Bar: ۵ LED را پله‌ای روشن کنید.
  • ۱۰پروژه ۱۰ — RGB: رنگ‌های مختلف را با LED RGB نمایش دهید.
