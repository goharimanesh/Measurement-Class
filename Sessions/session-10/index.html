⏱️ اصل کار — زمان پرواز

۱. ارسال

سنسور یک پالس ۱۰µs به پین Trig می‌فرستد و ۸ پالس ۴۰kHz منتشر می‌کند.

۲. دریافت

موج به جسم برخورد کرده و برمی‌گردد. پین Echo تا زمان دریافت HIGH می‌ماند.

۳. محاسبه

فرمول: distance = duration × 0.034 / 2

cm = duration (µs) / 58
inch = duration (µs) / 148

⚠️ محدودیت‌ها و نکات مهم

سطوح جاذب

پارچه، فوم، پشم شیشه موج را جذب می‌کنند و اندازه‌گیری خطا می‌دهد.

حداقل فاصله

کمتر از ۲cm قابل اندازه‌گیری نیست (محدودیت فیزیکی سنسور).

تأخیر بین اندازه‌گیری

حداقل ۶۰ms بین دو اندازه‌گیری متوالی برای جلوگیری از تداخل.

اثر دما

سرعت صوت با دما تغییر می‌کند (۰.۶ m/s به ازای هر °C).

زاویه ۱۵°

مخروط پخش موج — اجسام باریک ممکن است تشخیص داده نشوند.

فیلتر نرم‌افزاری

میانگین‌گیری ۳ تا ۵ نمونه برای کاهش نویز.

🔧 نکات اتصال و تجربه‌ها

۱. تغذیه ۵V

HC-SR04 با ۵V کار می‌کند. برای ESP32 (۳.۳V) از تقسیم مقاومتی برای Echo استفاده کنید.

۲. تقسیم مقاومتی

برای ESP32: دو مقاومت ۱kΩ و ۲kΩ برای کاهش ۵V به ۳.۳V.

۳. نویز

کابل شیلددار + خازن ۰.۱µF نزدیک سنسور. دور از موتور/رله.

۴. فیلتر نرم‌افزاری

میانگین‌گیری ۳-۵ نمونه. حذف مقادیر پرت (outliers).

۵. تأخیر ۶۰ms

حداقل ۶۰ms بین دو اندازه‌گیری متوالی.

۶. مکان‌یابی

فاصله روی Serial Monitor

ساده‌ترین پروژه: خواندن فاصله و نمایش روی سریال. پایه‌ای برای همه پروژه‌های بعدی.

• VCC → 5V

باز کردن Wokwi
/* * Project 1: Basic Distance Measurement on Serial Monitor * Designed by Dr. Masoud Goharimanesh * Course: Measurement Systems and Control * Session 10: Ultrasonic Distance Sensor (HC-SR04) */ const int TRIG_PIN = 9; const int ECHO_PIN = 10; void setup() { pinMode(TRIG_PIN, OUTPUT); pinMode(ECHO_PIN, INPUT); Serial.begin(9600); Serial.println("HC-SR04 - Dr. Goharimanesh"); } void loop() { // Send 10us trigger pulse digitalWrite(TRIG_PIN, LOW); delayMicroseconds(2); digitalWrite(TRIG_PIN, HIGH); delayMicroseconds(10); digitalWrite(TRIG_PIN, LOW); // Read echo pulse duration long duration = pulseIn(ECHO_PIN, HIGH, 30000); // Calculate distance in cm float distance = duration * 0.0343 / 2.0; Serial.print("Distance: "); Serial.print(distance); Serial.println(" cm"); delay(200); }

/* * Project 5: Ultrasonic Radar with Servo and OLED * Designed by Dr. Masoud Goharimanesh * Course: Measurement Systems and Control * Session 10: Ultrasonic Distance Sensor (HC-SR04) */ #include <Servo.h> #include <Wire.h> #include <Adafruit_SSD1306.h> #define TRIG_PIN 9 #define ECHO_PIN 10 Servo radarServo; Adafruit_SSD1306 display(128, 64, &Wire, -1); int angle = 0; int step = 5; void setup() { pinMode(TRIG_PIN, OUTPUT); pinMode(ECHO_PIN, INPUT); radarServo.attach(3); display.begin(SSD1306_SWITCHCAPVCC, 0x3C); display.clearDisplay(); } float getDistance() { digitalWrite(TRIG_PIN, LOW); delayMicroseconds(2); digitalWrite(TRIG_PIN, HIGH); delayMicroseconds(10); digitalWrite(TRIG_PIN, LOW); long d = pulseIn(ECHO_PIN, HIGH, 30000); return d * 0.0343 / 2.0; } void loop() { radarServo.write(angle); delay(30); float dist = getDistance(); display.clearDisplay(); display.setTextSize(1); display.setCursor(0, 0); display.print("Angle: "); display.print(angle); display.setCursor(0, 56); display.print("Dist: "); display.print(dist, 1); display.print("cm"); display.setCursor(0, 48); display.print("Dr. Goharimanesh"); if (dist < 40) { int x = map(angle, 0, 180, 0, 127); int y = map(dist, 0, 40, 50, 10); display.fillCircle(x, y, 2, WHITE); } display.display(); angle += step; if (angle >= 180 || angle <= 0) step = -step; }

/* * Project 6: IoT Distance Monitoring via MQTT * Designed by Dr. Masoud Goharimanesh * Course: Measurement Systems and Control * Session 10: Ultrasonic Distance Sensor (HC-SR04) */ #include <WiFi.h> #include <PubSubClient.h> const char* ssid = "Wokwi-GUEST"; const char* password = ""; const char* mqtt_server = "broker.hivemq.com"; WiFiClient espClient; PubSubClient client(espClient); const int TRIG_PIN = 5; const int ECHO_PIN = 18; void setup_wifi() { WiFi.begin(ssid, password); while (WiFi.status() != WL_CONNECTED) { delay(500); } } void reconnect() { while (!client.connected()) { if (client.connect("ESP32Distance")) { } else { delay(2000); } } } void setup() { pinMode(TRIG_PIN, OUTPUT); pinMode(ECHO_PIN, INPUT); Serial.begin(115200); setup_wifi(); client.setServer(mqtt_server, 1883); } void loop() { if (!client.connected()) reconnect(); client.loop(); digitalWrite(TRIG_PIN, LOW); delayMicroseconds(2); digitalWrite(TRIG_PIN, HIGH); delayMicroseconds(10); digitalWrite(TRIG_PIN, LOW); long d = pulseIn(ECHO_PIN, HIGH, 30000); float dist = d * 0.0343 / 2.0; char msg[10]; dtostrf(dist, 4, 1, msg); client.publish("goharimanesh/distance", msg); delay(1000); }

/* * Project 7: Intrusion Detection with PIR and HC-SR04 * Designed by Dr. Masoud Goharimanesh * Course: Measurement Systems and Control * Session 10: Ultrasonic Distance Sensor (HC-SR04) */ #include <WiFi.h> #include <PubSubClient.h> const char* ssid = "Wokwi-GUEST"; const char* mqtt_server = "broker.hivemq.com"; WiFiClient espClient; PubSubClient client(espClient); const int TRIG_PIN = 5; const int ECHO_PIN = 18; const int PIR_PIN = 19; void setup() { pinMode(TRIG_PIN, OUTPUT); pinMode(ECHO_PIN, INPUT); pinMode(PIR_PIN, INPUT); Serial.begin(115200); WiFi.begin(ssid, ""); while (WiFi.status() != WL_CONNECTED) { delay(500); } client.setServer(mqtt_server, 1883); } void loop() { if (!client.connected()) { if (client.connect("ESP32Alarm")) { } else { delay(2000); return; } } digitalWrite(TRIG_PIN, LOW); delayMicroseconds(2); digitalWrite(TRIG_PIN, HIGH); delayMicroseconds(10); digitalWrite(TRIG_PIN, LOW); long d = pulseIn(ECHO_PIN, HIGH, 30000); float dist = d * 0.0343 / 2.0; int motion = digitalRead(PIR_PIN); if (motion == HIGH && dist < 100.0) { client.publish("goharimanesh/alarm", "INTRUSION"); Serial.println("Alarm!"); } delay(1000); }
